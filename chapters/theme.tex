% ---------------------------------------------------------------
%  Feuille de route : 5 étapes, l'étape courante est mise en avant
% ---------------------------------------------------------------
\newcommand{\etape}[1]{%
  \ifcase#1\or Deux systèmes\\qui traitent le langage%
  \or Comparer\\le modèle au cerveau%
  \or Ouvrir la boîte :\\langage, mémoire, pensée%
  \or Mon travail :\\des modèles plus humains ?%
  \or Application :\\décoder le langage\fi}
\newcommand{\roadmap}[1]{%
  \begin{tikzpicture}[
      st/.style={circle, minimum size=0.95cm, font=\bfseries\large, inner sep=0pt},
      lb/.style={text width=2.3cm, align=center, font=\scriptsize, anchor=north}]
    \draw[gris, line width=1.2pt] (0,0) -- (11.2,0);
    \foreach \i in {1,...,5}{%
      \pgfmathsetmacro{\x}{(\i-1)*2.8}
      \ifnum\i=#1
        \node[st, fill=jaune, text=fond] (s\i) at (\x,0) {\i};
        \node[lb, text=jaune, font=\scriptsize\bfseries] at (\x,-0.65) {\etape{\i}};
      \else
        \ifnum\i<#1
          \node[st, fill=fondclair, draw=jaune, text=jaune] (s\i) at (\x,0) {\i};
          \node[lb, text=gris] at (\x,-0.65) {\etape{\i}};
        \else
          \node[st, fill=fondclair, draw=gris, text=gris] (s\i) at (\x,0) {\i};
          \node[lb, text=gris] at (\x,-0.65) {\etape{\i}};
        \fi
      \fi}
  \end{tikzpicture}}
