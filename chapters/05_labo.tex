\begin{frame}{Ma question de recherche}
\begin{block}{Rendre les modèles plus humains aide-t-il ?}
    Deux idées pour rapprocher un modèle de langage de l'humain.
\end{block}
\begin{itemize}
    \item \textbf{Contrainte de mémoire} : une architecture qui limite le contexte accessible, comme la mémoire de travail humaine
    \item \textbf{Contrainte développementale} : un entraînement sur 10 à 100 millions de mots, ce qu'entend un enfant (BabyLM)
\end{itemize}
\vspace{0.5em}
Ces contraintes rapprochent-elles les modèles du \textbf{comportement} humain et du \textbf{cerveau} ?
\src{Prolonge : Hosseini et al. 2024 ; Kuribayashi et al. 2021 ; Michaelov et al. 2024 ; AlKhamissi et al. 2025}
\end{frame}

    \begin{itemize}

\begin{frame}{La méthode : comparer des géométries}
\begin{enumerate}
    \item Pour un même ensemble de stimuli, on construit une matrice de distances (RDM) pour le modèle et pour l'humain
    \item On mesure combien la RDM du modèle améliore la prédiction de la RDM humaine, \textbf{au-delà de la longueur et de la fréquence des mots}
    \item Le score est normalisé par un plafond de bruit
\end{enumerate}
\vspace{0.5em}
\begin{itemize}
    \item \textbf{Similarité comportementale} : surprise du modèle contre temps de lecture
    \item \textbf{Similarité cérébrale} : embeddings contre signaux cérébraux
\end{itemize}
\end{frame}

\begin{frame}{Résultat 1 : deux cibles qui se dissocient}
\begin{itemize}
    \item \textbf{Aucun modèle n'est le meilleur sur les deux}
    \item Les modèles BabyLM et à mémoire limitée sont meilleurs côté \textbf{comportement}
    \item Les grands LLM standards sont meilleurs côté \textbf{cerveau}
    \item À taille et perplexité égales, les contraintes vont avec plus de similarité comportementale mais moins de similarité cérébrale
\end{itemize}

\end{frame}

\begin{frame}{Résultat 2 : la mémoire n'explique pas l'avantage}
\begin{itemize}
    \item Mots qui sollicitent la mémoire : ceux qui ferment une dépendance syntaxique longue
    \item Le signal humain est plus fort sur ces mots (temps de lecture, activité intracrânienne)
    \item Mais la contrainte de mémoire \textbf{n'apporte pas d'avantage spécifique} sur ces mots
\end{itemize}
\begin{block}{Conclusion}
    L'avantage comportemental des contraintes ne vient pas de limites de mémoire partagées avec l'humain.
\end{block}
\end{frame}

\begin{frame}{Ce qu'il faut retenir}
\begin{itemize}
    \item Ce qui rend la \textbf{difficulté de traitement} humaine n'est pas ce qui rend les \textbf{représentations} proches du cerveau
    \item L'alignement humain doit être évalué \textbf{sur les deux}
    \item Similarité de géométrie ne prouve pas un algorithme commun
\end{itemize}
\vspace{0.5em}
\textbf{Limites} : familles de modèles qui diffèrent par plus d'un facteur, données surtout en anglais.
\end{frame}
